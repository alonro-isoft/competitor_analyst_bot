        function displayAnalysis(data, mySolution, competitorSolution) {

            let html = '';

           

            // הוספת פיסקת הסיכום

            if (data.summary) {

                html += `<p>${data.summary}</p>`;

                html += '</div>';

            }

            // טבלת השוואת יכולות

            if (data.capabilities && data.capabilities.length > 0) {

                html += '<div class="table-container mb-8">';

                html += `<h2 class="text-2xl font-bold mb-4">השוואת יכולות</h2>`;

                html += '<table>';

                html += `<thead><tr><th>יכולת</th><th>${mySolution}</th><th>${competitorSolution}</th></tr></thead>`;

                html += '<tbody>';

                data.capabilities.forEach(item => {

                    html += `<tr><td>${item.feature}</td><td>${item.mySolution}</td><td>${item.competitorSolution}</td></tr>`;

                });

                html += '</tbody></table>';

                html += '</div>';

            }

            // טבלת יתרונות וחסרונות

            if (data.prosAndCons && data.prosAndCons.length > 0) {

                html += '<div class="table-container">';

                html += `<h2 class="text-2xl font-bold mb-4">יתרונות וחסרונות</h2>`;

                html += `<thead><tr><th>היבט</th><th>${mySolution}</th><th>${competitorSolution}</th></tr></thead>`;

                html += '<tbody>';

                data.prosAndCons.forEach(item => {

                    html += `<tr><td>${item.aspect}</td><td>${item.mySolution}</td><td>${item.competitorSolution}</td></tr>`;

                });

                html += '</div>';

            }

        }
